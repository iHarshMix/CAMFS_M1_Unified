\section{Formulas \& Equations}
% ==========================================

\subsection{Phase 1: Unimodal Encoder Alignment}

For client $i$, owned modality $m \in O(i)$, and spatial position $p$ with ground-truth class $c(p) \in \{0:\text{BG}, 1:\text{NCR}, 2:\text{ED}, 4:\text{ET}\}$, spatial feature embeddings $z_m(p)$ are aligned with class prototypes $\text{Proto}_m^c$ via location-wise InfoNCE loss:

\begin{equation}
\mathcal{L}_{\text{unimodal-align}}(z_m) = -\frac{1}{|P_m|} \sum_{p \in P_m} \log \frac{\exp\left(\text{sim}(z_m(p),\, \text{Proto}_m^{c(p)}) / \tau\right)}{\sum_{c' \in \{0,1,2,4\}} \exp\left(\text{sim}(z_m(p),\, \text{Proto}_m^{c'}) / \tau\right)}
\end{equation}

where $\text{sim}(u, v) = \frac{u^T v}{\|u\|_2 \|v\|_2}$ denotes cosine similarity.

Local prototype estimates for class $c \in \{0,1,2,4\}$ are computed as spatial centroid means:
\begin{equation}
\text{Proto}_{m,i}^{c,t} = \frac{1}{|P_{m,i}^c|} \sum_{p \in P_{m,i}^c} z_m(p)
\end{equation}

Federated prototype aggregation for client $k$ is evidence-weighted by pixel count $|P_{m,j}^c|$:
\begin{equation}
\text{Proto}_m^{c, t+1} \to k = \sum_{j:\, m \in \text{Send}(j),\, R_{\text{recv}}(k,j,m)=1} \frac{|P_{m,j}^c|}{N_m^{c,(k)}} \text{Proto}_{m,j}^{c,t}
\end{equation}

\subsection{Phase 1 Stopping Criterion \& Freeze Procedure}

Phase 1 completes after $T_1$ minimum rounds when global prototype drift across all classes $c \in \{0,1,2,4\}$ falls below threshold $\epsilon$ for $K$ consecutive rounds:

\begin{equation}
\frac{1}{4 C_{\text{cls}}} \sum_{m \in \mathcal{M}} \sum_{c \in \{0,1,2,4\}} \left\| \text{Proto}_m^{c,t} - \text{Proto}_m^{c,t-1} \right\|_2 < \epsilon \quad \text{for } K \text{ consecutive rounds}
\end{equation}

At transition, encoder weights are locked ($\theta_{E_m}^*$) and computation graph detachment is enforced:
\begin{equation}
z_m = \text{detach}\left(\text{Encoder}_m^*(x_m)\right) \implies \frac{\partial z_m}{\partial \theta_{E_m}^*} \equiv 0
\end{equation}

\subsection{Phase 2: Track-Isolated Fusion Training}

Phase 2 computes task segmentation loss (Dice + Cross-Entropy) and fused prototype alignment for ground-truth classes $c \in \{0,1,2,4\}$:

\begin{equation}
\mathcal{L}_{\text{task}}(y, \hat{y}) = \mathcal{L}_{\text{CE}}(y, \hat{y}) + \mathcal{L}_{\text{Dice}}(y, \hat{y})
\end{equation}

\begin{equation}
\mathcal{L}_{\text{fusion-align}}(z_{S'}) = -\frac{1}{|P_{S'}|} \sum_{p \in P_{S'}} \log \frac{\exp\left(\text{sim}(z_{S'}(p),\, \text{FusedProto}_{S'}^{c(p)}) / \tau\right)}{\sum_{c' \in \{0,1,2,4\}} \exp\left(\text{sim}(z_{S'}(p),\, \text{FusedProto}_{S'}^{c'}) / \tau\right)}
\end{equation}

\subsection{Send-Gated Track Aggregation Equations}

Track fusion updates are aggregated strictly over clients whose declared \textbf{Send policy} matches track identity $S'$:

\begin{equation}
\theta_{\text{FusionHead}_{S'}}^{t+1} \to k = \sum_{\substack{j:\, \text{Send}(j)=S',\\ R_{\text{send}}^{\text{track}}(j,S')=1,\\ R_{\text{recv}}^{\text{track}}(k,j,S')=1}} \frac{n_j}{N_{S'}^{(k)}} \theta_{\text{FusionHead}_{S'}, j}^t
\end{equation}

\section{Consolidated Loss Functions}
% ==========================================

\subsection{Phase 1 Overall Client Loss}
\begin{equation}
\mathcal{L}_i^{\text{Phase 1}} = \lambda_1 \sum_{m \in O(i)} \mathcal{L}_{\text{unimodal-align}}(z_m)
\end{equation}

\subsection{Phase 2 Overall Client Loss}
\begin{equation}
\mathcal{L}_i^{\text{Phase 2}} = \mathcal{L}_{\text{task}}(y_i, \hat{y}_i) + \lambda_2 \mathcal{L}_{\text{fusion-align}}(z_{S'})
\end{equation}

Note: $\lambda_1$'s term is strictly absent in Phase 2 because encoder parameters $\theta_{E_m}^*$ are frozen ($\text{requires\_grad}=\text{False}$) and graph-detached.

\subsection{Ground-Truth Label Mapping}
BraTS NIfTI masks contain 4 ground-truth label classes $c \in \{0, 1, 2, 4\}$:
\begin{itemize}
    \item \textbf{Class $c=0$ (BG):} Background and healthy brain tissue.
    \item \textbf{Class $c=1$ (NCR/NET):} Necrotic and non-enhancing tumor core.
    \item \textbf{Class $c=2$ (ED):} Peritumoral edema.
    \item \textbf{Class $c=4$ (ET):} Enhancing tumor (Label 3 is unused).
\end{itemize}

\subsection{Evaluation Metrics Formulations}

\subsubsection{Dice Similarity Coefficient (DSC)}
For ground-truth binary mask $Y$ and predicted mask $\hat{Y}$ over region $R \in \{\text{WT}, \text{TC}, \text{ET}\}$:
\begin{equation}
\text{DSC}_R(Y, \hat{Y}) = \frac{2 |Y_R \cap \hat{Y}_R|}{|Y_R| + |\hat{Y}_R|} \times 100\%
\end{equation}

\subsubsection{Compliance-Performance Gap ($\Delta_{\text{compliance}}$)}
Measures the explicit accuracy cost of hard consent compliance relative to an unrestricted oracle:
\begin{equation}
\Delta_{\text{compliance}}^{(i)} = \text{DSC}_{\text{Oracle}}^{(i)} - \text{DSC}_{\text{CAMFS}}^{(i)}
\end{equation}

\subsubsection{Purity Probe Classification Accuracy}
A binary classifier $f_{\text{probe}}$ is trained on bottleneck embeddings $z_m$ or fused features $z_{S'}$ to predict exposure to disallowed modality $m^*$:
\begin{equation}
\text{Acc}_{\text{probe}} = \mathbb{P}\left[ f_{\text{probe}}(z) == \text{is\_exposed}(m^*) \right]
\end{equation}
Compliance invariant: $\text{Acc}_{\text{probe}} \equiv \text{Null Baseline (Chance Level)}$.
